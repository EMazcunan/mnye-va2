Tema 7: Algunas visualizaciones

En este documento encontrarás algunas visualizaciones que pueden resultarte útiles para interpretar los resultados de algunos problemas de los apuntes.

d3 = require("d3@7")
jstat = require("jstat@1.9.6")

Problema 7

La siguiente visualización permite comparar las probabilidades de que Ana llegue tarde para diferentes valores de la correlación entre \(X\) e \(Y\).

  • La gráfica gris con línea discontinua se corresponde con el caso de correlación \(0\), como en el apartado (a), y se mantendrá siempre fija como referencia.

  • La gráfica azul muestra inicialmente el caso de correlación \(0.5\), como en el apartado (b). Puedes utilizar el deslizador para modificar el valor de la correlación y ver cómo cambiaría el resultado del apartado (b).

viewof rho_dynamic = Inputs.range([-0.8,0.8], {value: 0.5, step: 0.1, label: "Correlación entre X e Y"})
viewof umbral = {
  // --- CONFIGURACIÓN DEL LIENZO ---
  const width = 700;
  const height = 400;
  const margin = {top: 40, right: 30, bottom: 40, left: 50}; 
  
  // Parámetros estadísticos
  const mu = 45;
  const sigma_fixed = 5;
  const sigma_dynamic = (25 + 24*rho_dynamic)**0.5;
  const x_min = 30;
  const x_max = 60;
  const x_domain = [x_min, x_max];
  const y_max = 0.18;
  const y_domain = [0, y_max];

  // Escalas de D3
  const x = d3.scaleLinear().domain(x_domain).range([margin.left, width - margin.right]);
  const y = d3.scaleLinear().domain(y_domain).range([height - margin.bottom, margin.top]);

  // Funciones Matemáticas
  const pdf = (val, m, s) => (1 / (s * Math.sqrt(2 * Math.PI))) * Math.exp(-0.5 * Math.pow((val - m) / s, 2));
  const data_range = d3.range(x_domain[0], x_domain[1], 0.1);

  // --- CREACIÓN DEL SVG ---
    const svg = d3.create("svg").attr("viewBox", [0, 0, width, height]).style("background", "white").style("border", "1px solid #eee");

  // Ejes
  svg.append("g").attr("transform", `translate(0,${height - margin.bottom})`).call(d3.axisBottom(x));
  svg.append("g").attr("transform", `translate(${margin.left},0)`).call(d3.axisLeft(y));

  // Título
  svg.append("text").attr("x", width/2).attr("y", 25).attr("text-anchor", "middle").style("font-weight", "bold").text("Comparativa de correlación 0 y la correlación elegida");

  // 1. PDF Fija y Variable
  svg.append("path").datum(data_range).attr("fill", "none").attr("stroke", "gray").attr("stroke-dasharray", "4,4").attr("d", d3.line().x(d => x(d)).y(d => y(pdf(d, mu, sigma_fixed))));
  svg.append("path").datum(data_range).attr("fill", "none").attr("stroke", "rgb(44, 133, 235)").attr("stroke-width", 2).attr("d", d3.line().x(d => x(d)).y(d => y(pdf(d, mu, sigma_dynamic))));

  // 2. Contenedores para áreas (cola derecha X > k)
  const shade_fixed_color = "rgba(128, 128, 128, 0.25)"
  const shade_fixed = svg.append("path").attr("fill", shade_fixed_color);
  const shade_dynamic_color = "rgba(44, 133, 235, 0.25)";
  const shade_dynamic = svg.append("path").attr("fill", shade_dynamic_color);

  // --- NUEVA LEYENDA DE PROBABILIDADES (X > k) ---
  const legend_x_left = width - margin.right - 140;
  const legend_y_top = (height - margin.bottom)/2  - 35;

  svg.append("rect")
    .attr("x", legend_x_left)      // Posición horizontal
    .attr("y", legend_y_top-13)      // Posición vertical
    .attr("width", 13)  // Ancho
    .attr("height", 13) // Alto
    .style("fill", shade_fixed_color);
  const legend_fixed = svg.append("text")
    .attr("x", legend_x_left + 20)
    .attr("y", legend_y_top)
    //.attr("fill", "gray")
    .style("font-size", "13px")
    .style("font-weight", "500");

  svg.append("rect")
    .attr("x", legend_x_left)      // Posición horizontal
    .attr("y", legend_y_top -13 + 20)      // Posición vertical
    .attr("width", 13)  // Ancho
    .attr("height", 13) // Alto
    .style("fill", shade_dynamic_color);
  const legend_dynamic = svg.append("text")
    .attr("x", legend_x_left + 20)
    .attr("y", legend_y_top + 20)
    //.attr("fill", "blue")
    .style("font-size", "13px")
    .style("font-weight", "500");

  // --- LÓGICA DE ACTUALIZACIÓN ---
  function update(val_umbral){
    const range_shade = d3.range(val_umbral, x_domain[1], 0.1);
    
    // Actualizar áreas sombreadas
    shade_fixed.datum(range_shade).attr("d", d3.area().x(d => x(d)).y0(y(0)).y1(d => y(pdf(d, mu, sigma_fixed))));
    shade_dynamic.datum(range_shade).attr("d", d3.area().x(d => x(d)).y0(y(0)).y1(d => y(pdf(d, mu, sigma_dynamic))));
    
    // Actualizar posición de la línea roja y etiqueta k
    handle.attr("transform", `translate(${x(val_umbral)},0)`);
    label.text(`t = ${val_umbral.toFixed(1)}`);

    // --- ACTUALIZAR LEYENDA (Calculando P(X < k)) ---
    const p_fixed = 1-jstat.normal.cdf(val_umbral, mu, sigma_fixed);
    const p_dynamic = 1-jstat.normal.cdf(val_umbral, mu, sigma_dynamic);

    legend_fixed.text(`${p_fixed.toFixed(4)}`);
    legend_dynamic.text(`${p_dynamic.toFixed(4)}`);
    
    // Emitir valor al documento
    svg.node().value = val_umbral;
    svg.node().dispatchEvent(new CustomEvent("input"));
  }

  // --- COMPONENTE ARRASTRABLE ---
  let current_umbral = 50.0;
  const drag = d3.drag().on("drag", (event) => {
    const rawX = x.invert(event.x);
    current_umbral = Math.max(x_domain[0], Math.min(x_domain[1], rawX));
    update(current_umbral);
  });

  const handle_color = "rgba(243, 133, 110, 1)";
  const handle = svg.append("g").attr("cursor", "ew-resize").call(drag);
  handle.append("line").attr("y1", margin.top + 20).attr("y2", height - margin.bottom).attr("stroke", handle_color).attr("stroke-width", 2);
  handle.append("rect").attr("x", -10).attr("y", margin.top).attr("width", 20).attr("height", height - margin.top - margin.bottom).attr("fill", "transparent");
  const label = handle.append("text").attr("y", margin.top + 10).attr("text-anchor", "middle").attr("fill", handle_color).style("font-size", "12px").style("font-weight", "bold");

  update(current_umbral);
  return svg.node();
}
